-- Apply with the antivirus release, after existing attachments have been scanned.
-- Preserve all files; only the authorized API can release scanned bytes.
begin;
drop policy if exists remodel_files_server_only on storage.objects;
create policy remodel_files_server_only on storage.objects
  as restrictive for all to anon, authenticated
  using (bucket_id <> 'remodel-files')
  with check (bucket_id <> 'remodel-files');
update storage.buckets
set public = false, allowed_mime_types = array[
  'image/jpeg','image/png','image/webp','application/pdf','text/plain'
]
where id = 'remodel-files';
notify pgrst, 'reload schema';
commit;
