import { config } from "dotenv";
import { scanWithCloudmersive, AntivirusError } from "../lib/antivirus-client";
config({ path: ".env.local", quiet: true });

// The harmless industry-standard antivirus test string, never executable malware.
const eicar = [
  "X5O!P%@AP[4\\PZX54(P^)7CC)7}$",
  "EICAR-STANDARD-ANTIVIRUS-TEST-FILE!$H+H*",
].join("");
const options = { apiKey: process.env.CLOUDMERSIVE_API_KEY };
await scanWithCloudmersive(
  Buffer.from("Premium Remodel scanner connection test."),
  "text/plain",
  "txt",
  options,
);
await new Promise((resolve) => setTimeout(resolve, 1100));
let detected = false;
try {
  await scanWithCloudmersive(Buffer.from(eicar), "text/plain", "txt", options);
} catch (error) {
  if (error instanceof AntivirusError && error.reason === "blocked")
    detected = true;
  else throw error;
}
if (!detected)
  throw new Error(
    "Antivirus verification FAILED: test threat was not blocked. Do not activate uploads.",
  );
console.log(
  "Live antivirus verified: clean file accepted and EICAR test blocked.",
);
